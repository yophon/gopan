# 04 · 数据模型

PostgreSQL,goose 迁移。所有 id 用 UUID v7(时间有序,索引友好)。时间戳一律 `timestamptz`。

```sql
-- 用户
CREATE TABLE users (
    id            uuid PRIMARY KEY,
    username      text NOT NULL UNIQUE,
    password_hash text NOT NULL,              -- bcrypt
    quota_bytes   bigint NOT NULL DEFAULT 107374182400,  -- 100GB
    used_bytes    bigint NOT NULL DEFAULT 0,  -- 逻辑用量,秒传也计入
    is_admin      boolean NOT NULL DEFAULT false,
    created_at    timestamptz NOT NULL DEFAULT now()
);

-- 物理文件,全局按内容去重
CREATE TABLE blobs (
    id         uuid PRIMARY KEY,
    sha256     text NOT NULL UNIQUE,
    size       bigint NOT NULL,
    mime       text NOT NULL DEFAULT 'application/octet-stream',
    ref_count  int NOT NULL DEFAULT 0,
    verified   boolean NOT NULL DEFAULT false, -- 服务端重算 hash 通过后才可秒传命中
    media_meta jsonb,                          -- ffprobe 结果:时长、分辨率等
    created_at timestamptz NOT NULL DEFAULT now(),
    deref_at   timestamptz                     -- ref_count 归零时间,GC 宽限期起点
);
CREATE INDEX idx_blobs_gc ON blobs (deref_at) WHERE ref_count = 0;

-- 目录树节点
CREATE TABLE nodes (
    id         uuid PRIMARY KEY,
    owner_id   uuid NOT NULL REFERENCES users(id),
    parent_id  uuid REFERENCES nodes(id),      -- NULL = 用户根目录
    name       text NOT NULL,
    kind       text NOT NULL CHECK (kind IN ('file', 'folder')),
    blob_id    uuid REFERENCES blobs(id),      -- 仅 file 非空
    deleted_at timestamptz,                    -- 软删,回收站
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now()
);
-- 活跃节点同层同名唯一
CREATE UNIQUE INDEX idx_nodes_sibling
    ON nodes (owner_id, parent_id, name) WHERE deleted_at IS NULL;
CREATE INDEX idx_nodes_parent ON nodes (parent_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_nodes_trash  ON nodes (owner_id, deleted_at) WHERE deleted_at IS NOT NULL;
-- 文件名搜索
CREATE EXTENSION IF NOT EXISTS pg_trgm;
CREATE INDEX idx_nodes_name_trgm ON nodes USING gin (name gin_trgm_ops);

-- 分片上传会话
CREATE TABLE upload_sessions (
    id              uuid PRIMARY KEY,
    owner_id        uuid NOT NULL REFERENCES users(id),
    sha256          text NOT NULL,
    size            bigint NOT NULL,
    target_parent   uuid NOT NULL,
    target_name     text NOT NULL,
    minio_upload_id text NOT NULL,             -- S3 multipart uploadId
    part_size       int NOT NULL DEFAULT 16777216,
    status          text NOT NULL DEFAULT 'uploading'
        CHECK (status IN ('uploading','completing','verifying','done','failed','aborted')),
    fail_reason     text,
    created_at      timestamptz NOT NULL DEFAULT now(),
    expires_at      timestamptz NOT NULL       -- created_at + 48h,过期 worker Abort 清理
);
CREATE INDEX idx_sessions_cleanup ON upload_sessions (expires_at)
    WHERE status = 'uploading';
-- 已传分片不建表:实时问 MinIO ListObjectParts,它是唯一事实源

-- 分享
CREATE TABLE shares (
    id            uuid PRIMARY KEY,
    token         text NOT NULL UNIQUE,        -- 10 位 base58
    node_id       uuid NOT NULL REFERENCES nodes(id),
    created_by    uuid NOT NULL REFERENCES users(id),
    password_hash text,                        -- 可空;bcrypt
    expires_at    timestamptz,                 -- 可空 = 永久
    revoked_at    timestamptz,
    created_at    timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX idx_shares_owner ON shares (created_by) WHERE revoked_at IS NULL;

-- refresh token(双 token 的服务端半边)
CREATE TABLE refresh_tokens (
    id         uuid PRIMARY KEY,
    user_id    uuid NOT NULL REFERENCES users(id),
    token_hash text NOT NULL UNIQUE,           -- sha256(token),明文不落库
    family_id  uuid NOT NULL,                  -- 旋转链;重用检测吊销整族
    used_at    timestamptz,                    -- 非空 = 已旋转;再次出现即泄露
    revoked_at timestamptz,
    expires_at timestamptz NOT NULL,
    created_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX idx_refresh_family ON refresh_tokens (family_id);

-- 异步任务(worker pool 的持久化状态)
CREATE TABLE tasks (
    id         uuid PRIMARY KEY,
    kind       text NOT NULL
        CHECK (kind IN ('verify_hash','thumb','video_cover','office_pdf','media_probe')),
    blob_id    uuid NOT NULL REFERENCES blobs(id),
    status     text NOT NULL DEFAULT 'pending'
        CHECK (status IN ('pending','running','done','failed')),
    attempts   int NOT NULL DEFAULT 0,         -- 最多 3 次
    last_error text,
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),
    UNIQUE (kind, blob_id)                     -- 幂等:同一产物只派生一次
);
CREATE INDEX idx_tasks_pending ON tasks (status, created_at)
    WHERE status IN ('pending','running');

-- 派生产物(预览缓存索引)
CREATE TABLE derivatives (
    blob_id    uuid NOT NULL REFERENCES blobs(id),
    kind       text NOT NULL CHECK (kind IN ('thumb256','thumb2048','pdf','cover')),
    minio_key  text NOT NULL,                  -- derived/{sha256}/{kind}
    size       bigint NOT NULL,
    created_at timestamptz NOT NULL DEFAULT now(),
    PRIMARY KEY (blob_id, kind)
);
```

## 关键约束与不变量

- **blobs.ref_count 只在事务里变**:建 node +1、彻底删除 -1、复制引用 +1。归零时写 `deref_at`,再被引用时清空。
- **used_bytes 与 node 增删同事务**:秒传也加(逻辑记账),还原回收站不变(软删不退配额,彻底删除才退)。
- **子树查询统一用递归 CTE**,两处用:移动防成环、访客鉴权(node 是否在分享根下)。sqlc 里落成 `IsDescendant(ancestor, node)` 一个查询。
- **upload_sessions 不存分片明细**:MinIO 的 ListObjectParts 是唯一事实源,避免两边对账。
- 回收站里的文件夹整树跟着 deleted_at 标记(删除时一条 UPDATE 子树 CTE 完成),避免"父进了回收站子还活着"的裂缝。

## 容量估算

100 万 node 行 ≈ 300MB 含索引;blobs/tasks/derivatives 同量级。单机 Postgres 毫无压力,不需要分表设计。
